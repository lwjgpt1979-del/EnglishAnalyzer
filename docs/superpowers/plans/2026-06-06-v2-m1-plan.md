  ```python
  """译林版种子数据脚本（D-079 / M1）。

  使用 curriculum_ai_service dev mock 生成知识点骨架，
  通过 curriculum_service.persist_unit 写入数据库。
  无需真实 DeepSeek key。

  用法：
      cd backend
      DATABASE_URL=... python -m scripts.seed_yilin
  """
  from __future__ import annotations

  import asyncio
  import sys
  from pathlib import Path

  # 将 backend/ 加入 sys.path
  sys.path.insert(0, str(Path(__file__).parent.parent))

  from sqlalchemy.ext.asyncio import AsyncSession

  from app.core.database import async_session_factory
  from app.services.curriculum_ai_service import generate_unit
  from app.services.curriculum_service import persist_unit

  TEXTBOOKS = [
      {"textbook_version": "译林版", "grade": "小学5年级"},
      {"textbook_version": "译林版", "grade": "初中7年级"},
  ]
  SEMESTERS = ["上", "下"]
  UNITS_PER_SEMESTER = 6  # 每学期 6 个单元（译林版标准编排）


  async def seed_grade(
      db: AsyncSession,
      *,
      textbook_version: str,
      grade: str,
      units_per_semester: int = UNITS_PER_SEMESTER,
  ) -> None:
      """为一个教材-年级的上下两学期生成种子单元。"""
      for semester in SEMESTERS:
          for unit_no in range(1, units_per_semester + 1):
              ai_unit = await generate_unit(
                  textbook_version=textbook_version,
                  grade=grade,
                  semester=semester,
                  unit_no=unit_no,
              )
              cu = await persist_unit(db, ai_unit=ai_unit, content_status="published")
              print(f"  ✓ {textbook_version} {grade} {semester}学期 Unit{unit_no}: {cu.unit_title}")
      await db.commit()


  async def main() -> None:
      print("开始生成译林版种子数据（dev mock 模式）…")
      async with async_session_factory() as db:
          for tb in TEXTBOOKS:
              print(f"\n► {tb['textbook_version']} {tb['grade']}")
              await seed_grade(db, **tb)
      print("\n✅ 种子数据生成完成")


  if __name__ == "__main__":
      asyncio.run(main())
  ```

  ```vue
  <!-- V2 学期购买页 -->
  <template>
    <view class="page">
      <view class="header">
        <text class="title">解锁学期会员</text>
        <text class="sub">{{ textbook }} · {{ grade }} · {{ semester }}学期</text>
      </view>

      <!-- 档位选择 -->
      <view class="section">
        <text class="section-title">选择档位</text>
        <view class="tier-list">
          <view
            v-for="t in tiers"
            :key="t.key"
            class="tier-card"
            :class="{ selected: selectedTier === t.key }"
            @tap="selectedTier = t.key"
          >
            <text class="tier-name">{{ t.label }}</text>
            <text class="tier-desc">{{ t.desc }}</text>
            <text class="tier-price">¥{{ pricing[t.key] }}/学期</text>
          </view>
        </view>
      </view>

      <!-- 学期数选择 -->
      <view class="section">
        <text class="section-title">购买学期数</text>
        <view class="count-row">
          <view
            v-for="n in [1, 2, 3]"
            :key="n"
            class="count-btn"
            :class="{ selected: semesterCount === n }"
            @tap="semesterCount = n"
          >
            <text>{{ n }} 学期</text>
            <text class="count-sub">{{ n * 6 }}个月</text>
          </view>
        </view>
      </view>

      <!-- 总价 -->
      <view class="total-row">
        <text class="total-label">合计</text>
        <text class="total-price">¥{{ totalPrice }}</text>
      </view>

      <button class="btn-pay" :disabled="paying" @tap="onPay">
        {{ paying ? '处理中…' : '微信支付' }}
      </button>
    </view>
  </template>

  <script setup lang="ts">
  import { computed, onMounted, reactive, ref } from 'vue'
  import { onLoad } from '@dcloudio/uni-app'
  import { createOrder, payOrder } from '@/api/orders'
  import type { SemesterItem } from '@/types/api'

  const textbook = ref('译林版')
  const grade = ref('小学5年级')
  const semester = ref<'上' | '下'>('上')
  const selectedTier = ref('basic')
  const semesterCount = ref(1)
  const paying = ref(false)

  const pricing = reactive({ basic: 39, pro: 79, promax: 159 })

  const tiers = [
    { key: 'basic', label: 'Basic', desc: '知识点浏览 + 基础练习' },
    { key: 'pro', label: 'Pro', desc: '全功能 + 学情报告' },
    { key: 'promax', label: 'ProMax', desc: '全功能 + 作文精修 + AI 出题' },

  const totalPrice = computed(() => pricing[selectedTier.value as keyof typeof pricing] * semesterCount.value)

  onLoad((q: any) => {
    if (q?.textbook) textbook.value = decodeURIComponent(q.textbook)
    if (q?.grade) grade.value = decodeURIComponent(q.grade)
    if (q?.semester) semester.value = decodeURIComponent(q.semester) as '上' | '下'
  })

  async function onPay() {
    paying.value = true
    try {
      const semesters: SemesterItem[] = Array.from({ length: semesterCount.value }, () => ({
        textbook_version: textbook.value,
        grade: grade.value,
        semester: semester.value,
      }))

      const order = await createOrder({
        tier: selectedTier.value,
        order_type: 'new',
        semesters,
      })

      const payParams = await payOrder(order.id)

      // 唤起微信支付
      uni.requestPayment({
        ...payParams,
        success: () => {
          uni.showToast({ title: '购买成功！', icon: 'success' })
          setTimeout(() => uni.navigateBack(), 1500)
        },
        fail: (err: any) => {
          if (err.errMsg?.includes('cancel')) return
          uni.showToast({ title: '支付失败，请重试', icon: 'none' })
        },
      })
    } catch (e: any) {
      uni.showToast({ title: e?.message || '创建订单失败', icon: 'none' })
    } finally {
      paying.value = false
    }
  }
  </script>
  ```
